-- SISUD.VW_MULTA_COERCITIVA source

CREATE OR REPLACE VIEW SISUD.VW_MULTA_COERCITIVA
(NUMERO_EXPEDIENTE,ADMINISTRADO,RESOLUCION,FECHA_EMISION,NUMERO_REGISTRO,ESTADO_RESOLUCION,MEDIDA_ADMINISTRATIVA,CUM,CAM,MONTO_MULTA,MONTO_MULTA_REC,MONTO_MULTA_TFA,ESTADO_MULTA)
AS
SELECT DM.TXNUMEXP AS NUMERO_EXPEDIENTE,
    (
        SELECT LISTAGG(UPPER(AD.RAZONSOCIAL),';') WITHIN GROUP (ORDER BY AD.RAZONSOCIAL)
        FROM SISUD.T_MVD_MULTA_X_ADM MA
        INNER JOIN SISUD.TADMINISTRADO AD ON AD.IDADMINISTRADO = MA.IDADMINISTRADO
        WHERE MA.FGSITREG ='1' AND MA.IDMULTA = MD.IDMULTA
    ) AS ADMINISTRADO,
    CASE
        WHEN DM.TX_NUM_DOC = 'S/N' THEN DM.TX_NUM_DOC
        WHEN INSTR(DM.TX_NUM_DOC,'RESOLUCIÓN N°') > 0 THEN SUBSTR(DM.TX_NUM_DOC,15)
        WHEN INSTR(DM.TX_NUM_DOC,'RESOLUCIÓN DIRECTORAL N°') > 0 THEN SUBSTR(DM.TX_NUM_DOC,26)
        ELSE DM.TX_NUM_DOC
    END AS RESOLUCION,
    TO_CHAR(DM.FE_EMISION,'YYYY-MM-DD') AS FECHA_EMISION, DM.TX_NUM_REG AS NUMERO_REGISTRO,
    DECODE(DM.FGSITREG,'1','ACTIVO','0','INACTIVO') AS ESTADO_RESOLUCION,
    MD.TXMEDIDA_ADM AS MEDIDA_ADMINISTRATIVA, MD.TXCUM AS CUM, DM.TXCAM AS CAM,
    MD.NU_MULTA AS MONTO_MULTA, MD.NU_MULTA_RES_REC AS MONTO_MULTA_REC, MD.NU_MULTA_RES_TFA AS MONTO_MULTA_TFA,
    DECODE(MD.FGSITREG,'1','ACTIVO','0','INACTIVO') AS ESTADO_MULTA
FROM SISUD.T_MVD_DOCUMENTO_MULTA DM
INNER JOIN SISUD.T_MVD_MULTA_X_DOCUMENTO MD ON DM.IDDOCUMENTO_MULTA = MD.IDDOCUMENTO_MULTA_DS;
